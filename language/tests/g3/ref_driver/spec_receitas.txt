@@ arena | arena pool escopo alocador
# Arena e Pool de std:mem com alloc = a e alloc = scope
- mem.arena_create(n) faz um bloco de n bytes; new(T, alloc = a) aloca nele; mem.arena_destroy(a) solta tudo de uma vez.
- alloc = scope so vale como inicializador direto de let/var e solta no fim do bloco; nao sai dele (E0316). O nome antigo escopo da E0110: escreva scope.
- Pool so aceita new; delete(q, alloc = p) devolve o objeto; mem.pool_live(p) conta os vivos e mem.pool_destroy(p) fecha o pool.
import "std:mem"

struct Grifo {
    vida: i32 = 100
}

proc main() {
    let a = mem.arena_create(64)
    let g = new(Grifo, alloc = a)
    g.vida = 7
    let h = new(Grifo, alloc = a)
    print(g.vida + h.vida)
    mem.arena_destroy(a)
    let p = mem.pool_create()
    let q = new(Grifo, alloc = p)
    print(mem.pool_live(p))
    delete(q, alloc = p)
    print(mem.pool_live(p))
    mem.pool_destroy(p)
    let t = new(Grifo, alloc = scope)
    print(t.vida)
}

        return Erro.nao_achou

        return Erro.invalido
    }
    return id * 10
}

proc dobro(id: i32) -> i32 ! Erro {
    let v = try achar(id)
    return v * 2
}

    print(achar(3) or_else -1)
    print(achar(0) or_else -1)
    print(dobro(3) or_else 0)
    switch r in dobro(4) {
    case i32:
        print(r)
    case Erro:
        print("falhou")
    }
