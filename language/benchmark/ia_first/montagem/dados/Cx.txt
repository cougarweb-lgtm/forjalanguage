proc achar(a: []int, alvo: int) -> ?int {

            return i
        }
    }
    return none

    print(r + 1)

@c.ref

#define NADA (-1)

static int achar(const int *a, int n, int alvo) {
    for (int i = 0; i < n; i++) {

            return i;
        }
    }
    return NADA;
}

int main(void) {
    int v[3] = {5, 6, 7};
