@titulo Ordenar dois inteiros com qsort

@pedido
O programa ordena o array {2, 1} com qsort da biblioteca C, usando um procedimento da propria Forja para comparar, e imprime os dois elementos em ordem (1 e 2, um por linha).
@forja.fixture err_E0907_foreign_como_callback.fj
@forja.origem estratificado por codigo: E0907 (7 fixtures)
@forja.de

    var v: [2]i32 = {2, 1}
    qsort(&v[0] as rawptr, n = 2, tamanho = 4, comparar = labs)
}
@forja.para
proc comparar_i32(a: rawptr, b: rawptr) -> i32 {
    let x = a as ^i32
    let y = b as ^i32
    if x^ < y^ {
        return -1
    }
    if x^ > y^ {
        return 1
    }
    return 0
}

proc main() {
    var v: [2]i32 = {2, 1}
    qsort(&v[0] as rawptr, n = 2, tamanho = 4, comparar = comparar_i32)
    print(v[0])
    print(v[1])
}
