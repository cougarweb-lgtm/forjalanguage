@titulo Somar tres numeros criados (vazamento)
@classe vazamento
@pedido
O programa cria tres numeros no heap, soma e imprime o total (6) e tem de liberar tudo o que criou antes de terminar. Para o C, o programa conta os blocos vivos e termina com codigo 3 se sobrar algum.
@forja.decl {"fase": "run", "codigo": "R0009", "exit": 101}
@forja.ref
proc criar(v: int) -> ^int {
    let p = new(int, alloc = heap)
    p^ = v
    return p
}

proc main() {
    var total = 0
    for i in 1..=3 {
        let p = criar(i)
        total += p^
        delete(p)
    }
    print(total)
}
@forja.troca
        delete(p)
@forja.por
@c.decl {"fase": "run", "exit": 3}
@c.ref
#include <stdio.h>
#include <stdlib.h>

static int vivos = 0;

static int *criar(int v) {
    int *p = malloc(sizeof(int));
    if (p == NULL) {
        exit(1);
    }
    *p = v;
    vivos += 1;
    return p;
}

static void liberar(int *p) {
    free(p);
    vivos -= 1;
}

int main(void) {
    int total = 0;
    for (int i = 1; i <= 3; i++) {
        int *p = criar(i);
        total += *p;
        liberar(p);
    }
    printf("%d\n", total);
    if (vivos != 0) {
        fprintf(stderr, "vazamento: %d blocos vivos\n", vivos);
        return 3;
    }
    return 0;
}
