### DF11
@titulo Dividir ao meio um numero par
@classe R0003
@pedido
O procedimento dividir_ao_meio exige um numero par e devolve a metade. O programa imprime a metade de 8 (4) e a metade de 6 (3).
@forja.decl {"fase": "run", "codigo": "R0003", "exit": 101}
@forja.ref
proc dividir_ao_meio(n: int) -> int {
    assert(n % 2 == 0, msg = "n precisa ser par")
    return n / 2
}

proc main() {
    print(dividir_ao_meio(8))
    print(dividir_ao_meio(6))
}
@forja.troca
    print(dividir_ao_meio(6))
@forja.por
    print(dividir_ao_meio(7))
### DF12
@titulo Preparar com valor invalido
@classe R0004
@pedido
O procedimento preparar recusa numeros negativos. O main prepara o valor 3 e imprime pronto.
@forja.decl {"fase": "run", "codigo": "R0004", "exit": 1}
@forja.ref
enum Erro { falhou }

proc preparar(n: int) -> ! Erro {
    if n < 0 {
        return Erro.falhou
    }
}

proc main() -> ! Erro {
    try preparar(3)
    print("pronto")
}
@forja.troca
    try preparar(3)
@forja.por
    try preparar(0 - 3)
