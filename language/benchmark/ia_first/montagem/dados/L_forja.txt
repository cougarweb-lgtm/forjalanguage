        if i % 3 == 0 {

### L02 | recursao simples (fibonacci)
proc fib(n: int) -> int {
    if n < 2 {
        return n
    }
    return fib(n - 1) + fib(n - 2)
}

proc main() {
    for i in 0..<8 {
        print(fib(i))
    }
}
### L03 | copia de struct por valor x ponteiro
struct Ponto {
    x: int = 0
    y: int = 0
}

proc mover_copia(p: Ponto) -> int {
    var q = p
    q.x += 20
    return q.x
}

proc mover(p: ^Ponto) {
    p.x += 5
}

proc main() {
    var a = Ponto{x = 1, y = 2}
    let r = mover_copia(a)
    mover(&a)
    mover(&a)
    print(r)
    print(a.x)
    print(a.y)
}

### L05 | fatia e uma visao do array
proc soma(s: []int) -> int {
    var t = 0
    for x in s {
        t += x
    }
    return t
}

### L06 | enum e switch
enum Estado { parado, andando, correndo }

proc velocidade(e: Estado) -> int {
    switch e {
    case Estado.parado:
        return 0
    case Estado.andando:
        return 5
    case Estado.correndo:
        return 12
    }
    return -1
}

proc proximo(e: Estado) -> Estado {
    switch e {
    case Estado.parado:
        return Estado.andando
    case Estado.andando:
        return Estado.correndo
    case Estado.correndo:
        return Estado.parado
    }
    return Estado.parado
}

proc main() {
    var e = Estado.parado
    var total = 0
    for i in 0..<5 {
        let v = velocidade(e)
        print(v)
        total += v
        e = proximo(e)
    }
    print(total)
}
### L07 | erro explicito com ! e or_else
enum Erro { divisao_por_zero, negativo }

proc dividir(a: int, b: int) -> int ! Erro {
    if b == 0 {
        return Erro.divisao_por_zero
    }
    if a < 0 {
        return Erro.negativo
    }
    return a / b
}

proc main() {
    print(dividir(10, b = 2) or_else 100)
    print(dividir(10, b = 0) or_else 100)
    print(dividir(0 - 4, b = 2) or_else 200)
    switch r in dividir(7, b = 2) {
    case int:
        print(r)
    case Erro:
        print(99)
    }
}
### L08 | opcional com none
proc achar(a: []int, alvo: int) -> ?int {
    for i, x in a {
        if x == alvo {
            return i
        }
    }
    return none
}

proc main() {
    var v: [4]int = {4, 8, 15, 16}
    print(achar(v[..], alvo = 15) or_else 100)
    print(achar(v[..], alvo = 23) or_else 100)
    switch r in achar(v[..], alvo = 16) {
    case int:
        print(r)
    case none:
        print(0)
    }
}

### L10 | while e sequencia de Collatz
proc passos(n: int) -> int {
    var x = n
    var c = 0
    while x != 1 {
        if x % 2 == 0 {
            x = x / 2
        } else {
            x = 3 * x + 1
        }
        c += 1
    }
    return c
}

proc main() {
    print(passos(1))
    print(passos(6))
    print(passos(27))
}

### L14 | ordenacao por troca em fatia
proc ordenar(a: []int) -> int {
    let n = len(a)
    var trocas = 0
    for i in 0..<n {
        let lim = n - 1 - i
        for j in 0..<lim {
            if a[j] > a[j + 1] {
                let t = a[j]
                a[j] = a[j + 1]
                a[j + 1] = t
                trocas += 1
            }
        }
    }
    return trocas
}

proc main() {
    var v: [6]int = {5, 2, 9, 1, 7, 3}
    print(ordenar(v[..]))
    for x in v {
        print(x)
    }
}
### L15 | for x copia, for &x altera
proc main() {
    var a: [3]int = {1, 2, 3}
    for x in a {
        print(x * 2)
    }
    for &x in a {
        x^ = x^ * 10
    }
    var s = 0
    for i, x in a {
        s += x * (i + 1)
    }
    print(s)
    print(a[2])
}
### L16 | bits em inteiro sem sinal
proc contar_bits(n: u32) -> int {
    var x = n
    var c = 0
    while x != 0 {
        c += (x & 1) as int
        x = x >> 1
    }
    return c
}

proc main() {
    print(contar_bits(0xB7))
    print(contar_bits(255))
    let m: u32 = 0xF0
    print(((m >> 4) & 0x3) as int)
    print((m | 0x0F) as int)
}

    defer delete(l)

        append(&l, item = i * i)

        s += x
    }
    print(s)
    print(l[4])
}
### L18 | recursao com retorno antecipado
proc potencia(base: int, exp: int) -> int {
    if exp == 0 {
        return 1
    }
    let meio = potencia(base, exp = exp / 2)
    if exp % 2 == 0 {
        return meio * meio
    }
    return meio * meio * base
}

proc main() {
    print(potencia(2, exp = 10))
    print(potencia(3, exp = 5))
    print(potencia(7, exp = 0))
}
