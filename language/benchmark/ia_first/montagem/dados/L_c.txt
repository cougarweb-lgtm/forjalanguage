### L02 | recursao simples (fibonacci)
#include <stdio.h>

static int fib(int n) {
    if (n < 2) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}

int main(void) {
    for (int i = 0; i < 8; i++) {
        printf("%d\n", fib(i));
    }
    return 0;
}

### L05 | fatia e uma visao do array
#include <stdio.h>

typedef struct {
    int *dados;
    int tam;
} Fatia;

static int soma(Fatia s) {
    int t = 0;
    for (int i = 0; i < s.tam; i++) {
        t += s.dados[i];
    }
    return t;
}

int main(void) {
    int a[6] = {10, 20, 30, 40, 50, 60};
    Fatia meio = {&a[1], 3};
    printf("%d\n", meio.tam);
    a[2] = 0;
    printf("%d\n", soma(meio));
    Fatia tudo = {a, 6};
    printf("%d\n", soma(tudo));
    return 0;
}
### L06 | enum e switch
#include <stdio.h>

typedef enum { PARADO, ANDANDO, CORRENDO } Estado;

static int velocidade(Estado e) {
    switch (e) {
    case PARADO:
        return 0;
    case ANDANDO:
        return 5;
    case CORRENDO:
        return 12;
    }
    return -1;
}

static Estado proximo(Estado e) {
    switch (e) {
    case PARADO:
        return ANDANDO;
    case ANDANDO:
        return CORRENDO;
    case CORRENDO:
        return PARADO;
    }
    return PARADO;
}

int main(void) {
    Estado e = PARADO;
    int total = 0;
    for (int i = 0; i < 5; i++) {
        int v = velocidade(e);
        printf("%d\n", v);
        total += v;
        e = proximo(e);
    }
    printf("%d\n", total);
    return 0;
}
### L07 | erro explicito com ! e or_else
#include <stdio.h>

typedef enum { OK, DIVISAO_POR_ZERO, NEGATIVO } Erro;

static Erro dividir(int a, int b, int *resultado) {
    if (b == 0) {
        return DIVISAO_POR_ZERO;
    }
    if (a < 0) {
        return NEGATIVO;
    }
    *resultado = a / b;
    return OK;
}

static int dividir_ou(int a, int b, int padrao) {
    int r = 0;
    if (dividir(a, b, &r) != OK) {
        return padrao;
    }
    return r;
}

int main(void) {
    printf("%d\n", dividir_ou(10, 2, 100));
    printf("%d\n", dividir_ou(10, 0, 100));
    printf("%d\n", dividir_ou(0 - 4, 2, 200));
    int r = 0;
    if (dividir(7, 2, &r) == OK) {
        printf("%d\n", r);
    } else {
        printf("%d\n", 99);
    }
    return 0;
}
### L08 | opcional com none
#include <stdio.h>

#define NADA (-1)

static int achar(const int *a, int n, int alvo) {
    for (int i = 0; i < n; i++) {
        if (a[i] == alvo) {
            return i;
        }
    }
    return NADA;
}

static int ou_padrao(int r, int padrao) {
    return r == NADA ? padrao : r;
}

int main(void) {
    const int v[4] = {4, 8, 15, 16};
    printf("%d\n", ou_padrao(achar(v, 4, 15), 100));
    printf("%d\n", ou_padrao(achar(v, 4, 23), 100));
    int r = achar(v, 4, 16);
    if (r != NADA) {
        printf("%d\n", r);
    } else {
        printf("%d\n", 0);
    }
    return 0;
}

### L10 | while e sequencia de Collatz
#include <stdio.h>

static int passos(int n) {
    int x = n;
    int c = 0;
    while (x != 1) {
        if (x % 2 == 0) {
            x = x / 2;
        } else {
            x = 3 * x + 1;
        }
        c += 1;
    }
    return c;
}

int main(void) {
    printf("%d\n", passos(1));
    printf("%d\n", passos(6));
    printf("%d\n", passos(27));
    return 0;
}

### L14 | ordenacao por troca em fatia
#include <stdio.h>

static int ordenar(int *a, int n) {
    int trocas = 0;
    for (int i = 0; i < n; i++) {
        int lim = n - 1 - i;
        for (int j = 0; j < lim; j++) {
            if (a[j] > a[j + 1]) {
                int t = a[j];
                a[j] = a[j + 1];
                a[j + 1] = t;
                trocas += 1;
            }
        }
    }
    return trocas;
}

int main(void) {
    int v[6] = {5, 2, 9, 1, 7, 3};
    printf("%d\n", ordenar(v, 6));
    for (int i = 0; i < 6; i++) {
        printf("%d\n", v[i]);
    }
    return 0;
}

### L16 | bits em inteiro sem sinal
#include <stdio.h>
#include <stdint.h>

static int contar_bits(uint32_t n) {
    uint32_t x = n;
    int c = 0;
    while (x != 0) {
        c += (int)(x & 1);
        x = x >> 1;
    }
    return c;
}

int main(void) {
    printf("%d\n", contar_bits(0xB7));
    printf("%d\n", contar_bits(255));
    uint32_t m = 0xF0;
    printf("%d\n", (int)((m >> 4) & 0x3));
    printf("%d\n", (int)(m | 0x0F));
    return 0;
}
### L17 | lista dinamica que cresce
#include <stdio.h>
#include <stdlib.h>

typedef struct {
    int *dados;
    int tam;
    int cap;
} Lista;

static void anexar(Lista *l, int item) {
    if (l->tam == l->cap) {
        l->cap = l->cap * 2;
        l->dados = realloc(l->dados, (size_t)l->cap * sizeof(int));
        if (l->dados == NULL) {
            exit(1);
        }
    }
    l->dados[l->tam] = item;
    l->tam += 1;
}

int main(void) {
    Lista l = {malloc(2 * sizeof(int)), 0, 2};
    if (l.dados == NULL) {
        return 1;
    }
    for (int i = 1; i <= 5; i++) {
        anexar(&l, i * i);
    }
    printf("%d\n", l.tam);
    int s = 0;
    for (int i = 0; i < l.tam; i++) {
        s += l.dados[i];
    }
    printf("%d\n", s);
    printf("%d\n", l.dados[4]);
    free(l.dados);
    return 0;
}
### L18 | recursao com retorno antecipado
#include <stdio.h>

static int potencia(int base, int exp) {
    if (exp == 0) {
        return 1;
    }
    int meio = potencia(base, exp / 2);
    if (exp % 2 == 0) {
        return meio * meio;
    }
    return meio * meio * base;
}

int main(void) {
    printf("%d\n", potencia(2, 10));
    printf("%d\n", potencia(3, 5));
    printf("%d\n", potencia(7, 0));
    return 0;
}
